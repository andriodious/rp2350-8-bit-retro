cmake_minimum_required(VERSION 3.20)
list(APPEND PICO_BOARD_HEADER_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../src/boards/presto")
set(PICO_BOARD pimoroni_presto CACHE STRING "Presto diagnostic board")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "Generate UF2 in the build script")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(retro8bit_bridge_probe C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(retro8bit_bridge_probe main.c)
target_compile_options(retro8bit_bridge_probe PRIVATE -Wall -Wextra)
target_link_libraries(retro8bit_bridge_probe PRIVATE pico_stdlib hardware_i2c)
pico_enable_stdio_usb(retro8bit_bridge_probe 1)
pico_enable_stdio_uart(retro8bit_bridge_probe 0)
pico_add_extra_outputs(retro8bit_bridge_probe)
